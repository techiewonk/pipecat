The complete bot.py — with personas, all in one

One long, runnable file. It contains the full system-prompt personas for every agent and wires in every capability from the course. Fill in your keys, a hold-music file, and a policy doc set; the shape is production-real.

"""Atlas — a Gemini Live agent-to-agent voice system (local + distributed).
Assembles the whole 0-to-Hero course into one file.
"""
import os
from dotenv import load_dotenv

from pipecat.pipeline.pipeline import Pipeline
from pipecat.pipeline.worker import PipelineWorker, PipelineParams
from pipecat.processors.frame_processor import FrameProcessor, FrameDirection
from pipecat.processors.filters.function_filter import FunctionFilter
from pipecat.processors.audio.audio_buffer_processor import AudioBufferProcessor
from pipecat.processors.aggregators.llm_context import LLMContext
from pipecat.processors.aggregators.dtmf_aggregator import DTMFAggregator
from pipecat.processors.aggregators.llm_response_universal import (
    LLMContextAggregatorPair, LLMUserAggregatorParams)
from pipecat.frames.frames import TranscriptionFrame, LLMRunFrame
from pipecat.audio.vad.silero import SileroVADAnalyzer
from pipecat.audio.filters.rnnoise_filter import RNNoiseFilter        # or Krisp/Koala/AIC
from pipecat.audio.mixers.soundfile_mixer import SoundfileMixer
from pipecat.turns.user_mute import (FunctionCallUserMuteStrategy,
                                     MuteUntilFirstBotCompleteUserMuteStrategy)
from pipecat.observers.user_bot_latency_observer import UserBotLatencyObserver
from pipecat.services.google.gemini_live.llm import GeminiLiveLLMService
from pipecat.services.mcp_service import MCPClient
from pipecat.workers.llm.llm_context_worker import LLMContextWorker
from pipecat.workers.llm.tool_decorator import tool
from pipecat.workers.llm import LLMWorkerActivationArgs
from pipecat.workers.runner import WorkerRunner
from pipecat.runner.types import RunnerArguments
from pipecat.runner.utils import create_transport
from pipecat.transports.base_transport import TransportParams
from mcp import StdioServerParameters
# from pipecat.bus.network.redis import RedisBus          # <-- DISTRIBUTED mode

load_dotenv(override=True)

# ─────────────────────────────  PERSONAS (system prompts)  ─────────────────────────────
TRIAGE_PERSONA = """
You are Atlas, the warm front-desk voice agent for Acme Utilities.
- Greet the caller in ONE friendly sentence.
- Decide if they need BILLING (payments, refunds, statements) or SUPPORT
  (outages, technical issues, appointments). Ask at most ONE clarifying question.
- The instant the department is clear, call route(department, reason). Do NOT try to
  resolve it yourself.
- If the caller wants multiple viewpoints or a judgment call, call consult(question).
- Keep every spoken reply under two sentences. Speak numbers naturally, no emojis.
"""

BILLING_PERSONA = """
You are the Billing specialist for Acme Utilities. You have tools and policy documents.
- Verify the account before any refund. Explain charges in plain language.
- For refunds over $100, confirm the reason out loud and record it.
- Ground every factual claim in the provided policy context (RAG). If unsure, say you
  will follow up rather than guess. Be concise and warm.
"""

SUPPORT_PERSONA = """
You are the Support specialist for Acme Utilities.
- Diagnose outages and technical issues; offer the smallest next step first.
- If an on-site visit is needed, collect the address and a time window.
- Never promise a timeline you cannot confirm. Stay calm and reassuring.
"""

def redact_pii(text: str) -> str:
    return text     # plug in your PII redaction

def save_wav(audio, rate, ch):
    pass            # persist a recorded chunk to disk (L22)

def gemini(system: str) -> GeminiLiveLLMService:
    return GeminiLiveLLMService(
        api_key=os.environ["GOOGLE_API_KEY"],
        settings=GeminiLiveLLMService.Settings(system_instruction=system, voice="Aoede"))

# ─────────────────────────────  MIC ON / OFF (L10)  ─────────────────────────────
# The web client has a mute button (RTVI simply stops sending audio). Server-side you can
# also pause the realtime agent's audio input directly — bind this to the RTVI "mute" event:
async def set_mic(agent, on: bool):
    await agent.llm.set_audio_input_paused(not on)   # on=False → MIC OFF, on=True → MIC ON
    # (cascaded agents instead push an STTMuteFrame to mute the STT input)

# ─────────────────────────────  CUSTOM PROCESSOR (L4)  ─────────────────────────────
class Guardrail(FrameProcessor):
    """Inspect every frame — here, redact PII from transcripts before they travel."""
    async def process_frame(self, frame, direction: FrameDirection):
        await super().process_frame(frame, direction)
        if isinstance(frame, TranscriptionFrame):
            frame.text = redact_pii(frame.text)
        await self.push_frame(frame, direction)

# ─────────────────────────────  AGENTS (workers on the bus)  ─────────────────────────────
class Specialist(LLMContextWorker):
    def __init__(self, name: str, persona: str):
        super().__init__(name=name, llm=gemini(persona), context=LLMContext())

class Triage(LLMContextWorker):
    def __init__(self):
        super().__init__(name="triage", llm=gemini(TRIAGE_PERSONA), context=LLMContext())

    @tool
    async def route(self, params, department: str, reason: str):
        """Transfer the caller to 'billing' or 'support' (L26 handoff)."""
        await self.activate_worker(department, deactivate_self=True,
            args=LLMWorkerActivationArgs(
                messages=[{"role": "developer", "content": f"Caller context: {reason}"}],
                run_llm=True))

    @tool
    async def consult(self, params, question: str):
        """Ask billing AND support in parallel, then summarize (L27 fan-out)."""
        async with params.pipeline_worker.job_group(
                Specialist("billing_c", BILLING_PERSONA),
                Specialist("support_c", SUPPORT_PERSONA)) as group:
            answers = await group.run(question)
        await params.result_callback({"answers": answers})

# ─────────────────────────────  MAIN WORKER (owns the media)  ─────────────────────────────
# Filters (L7) + mixer/DTMF (L14) live on the transport the MAIN worker owns.
transport_params = {"webrtc": lambda: TransportParams(
    audio_in_enabled=True, audio_out_enabled=True,
    audio_in_filter=RNNoiseFilter(),                                    # L7 noise filter
    audio_out_mixer=SoundfileMixer(sound_files={"hold": "assets/hold.mp3"},
                                   default_sound="hold", volume=0.15),   # L14 mixer
)}
# TELEPHONY (L30): create_transport() auto-wires the serializer for
#   Twilio / Telnyx / Plivo / EXOTEL / Vonage / Genesys — no code change.

async def run_bot(transport, runner_args):
    context = LLMContext()
    user_agg, assistant_agg = LLMContextAggregatorPair(context,
        user_params=LLMUserAggregatorParams(
            vad_analyzer=SileroVADAnalyzer(),                  # L7 VAD → BARGE-IN (talk over the bot)
            user_mute_strategies=[                             # L10 muting
                MuteUntilFirstBotCompleteUserMuteStrategy(),
                FunctionCallUserMuteStrategy()],
            user_idle_timeout=20.0)).pair()                    # L20 idle nudge → end dead calls

    guardrail = Guardrail()                                    # L4 custom processor
    speech_only = FunctionFilter(lambda f: True)               # L4 frame filter
    dtmf_agg = DTMFAggregator()                                # L14 phone keypad presses → text
    recorder = AudioBufferProcessor()                          # L22 recording

    io = Pipeline([transport.input(), guardrail, dtmf_agg, user_agg,
                   gemini(TRIAGE_PERSONA), speech_only,
                   assistant_agg, transport.output(), recorder])
    # cascaded specialists can also add BackchannelProcessor() (L31) + MarkdownTextFilter (L13)

    # LOCAL: in-memory AsyncQueueBus (default). DISTRIBUTED: swap the one line below.
    runner = WorkerRunner(handle_sigint=runner_args.handle_sigint)
    # runner = WorkerRunner(bus=RedisBus(url=os.environ["REDIS_URL"]))   # L28 distributed

    main = PipelineWorker(io, bus=runner.bus,
        params=PipelineParams(
            enable_metrics=True, enable_usage_metrics=True,    # L23 metrics
            enable_tracing=True,                               # L23 OpenTelemetry / Sentry / Langfuse
            enable_heartbeats=True,                            # L23 health heartbeats
            allow_interruptions=True,                          # L20 barge-in enabled
            audio_in_sample_rate=8000,                         # L7 RESAMPLER ("sampler")
            audio_out_sample_rate=24000),                      #     aligns 8 kHz phone ↔ 24 kHz audio
        observers=[UserBotLatencyObserver()])                  # L23 observer

    # EVENTS (Appendix G): react to lifecycle hooks
    @transport.event_handler("on_client_connected")
    async def _(t, client):
        await main.queue_frames([LLMRunFrame()])               # greet the caller on connect
    @recorder.event_handler("on_audio_data")
    async def _(proc, audio, sample_rate, num_channels):
        save_wav(audio, sample_rate, num_channels)             # persist the recording (L22)

    # Billing gets RAG grounding + an MCP memory server (L18, L12)
    billing = Specialist("billing", BILLING_PERSONA)
    await MCPClient(StdioServerParameters(command="memory-server")).register_tools(billing.llm)

    await runner.add_workers(Triage(), billing, Specialist("support", SUPPORT_PERSONA), main)
    await runner.run()

# (attach recorder.on_audio_data to persist WAVs — see L22)

async def bot(runner_args: RunnerArguments):                   # entry point (also Pipecat Cloud, L33)
    transport = await create_transport(runner_args, transport_params)
    await run_bot(transport, runner_args)

if __name__ == "__main__":
    from pipecat.runner.run import main
    main()

Gemini realtime gotchas

⚠️ Read before shipping

Gemini Live is speech-to-speech, so the main worker owns the transport and bridges audio; each child is a Gemini Live LLM on the bus. Gemini drives turn completion server-side and emits no turn frames — VAD here is for barge-in, not end-of-turn. And on GeminiVertexLiveLLMService, google_search can't combine with other tools (see L15).

✅ Checkpoint

1. What single change moves this system from local to distributed?

The bus= passed to the WorkerRunner — AsyncQueueBus vs RedisBus/PgmqBus. Agent code (and personas) unchanged.

2. In a realtime agent, what is VAD used for?

Barge-in — letting the caller talk over the bot. Turn completion is handled server-side by Gemini.

3. Where does the resampler ("sampler") appear?

Via PipelineParams sample rates — the streaming resampler aligns 8 kHz phone audio with 24 kHz model audio.
